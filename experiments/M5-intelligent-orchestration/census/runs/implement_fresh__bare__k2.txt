def parse_duration(s):
    """Parse a duration string into total seconds.

    Examples: '45m' -> 2700, '2h' -> 7200, '1h30m' -> 5400, '90s' -> 90,
    '1h2m3s' -> 3723, '' -> 0. Surrounding whitespace is allowed.
    """
    s = s.strip()
    if not s:
        return 0

    total_seconds = 0
    time_parts = s.split('h')
    hours = int(time_parts[0]) if time_parts[0] else 0

    if len(time_parts) > 1:
        time_parts = time_parts[1].split('m')
        minutes = int(time_parts[0]) if time_parts[0] else 0
        total_seconds += hours * 3600 + minutes * 60

    if len(time_parts) > 1 and 'm' in time_parts[1]:
        time_parts = time_parts[1].split('s')
        seconds = int(time_parts[0]) if time_parts[0] else 0
        total_seconds += seconds

    return total_seconds
