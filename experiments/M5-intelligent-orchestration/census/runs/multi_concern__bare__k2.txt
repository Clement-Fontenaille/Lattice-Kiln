# graphlib.py
"""
Module for graph operations.
"""

from typing import List, Dict

def build_graph(edges: List[List[int]]) -> Dict[int, List[int]]:
    """
    Build a graph from a list of edges.

    Args:
        edges (List[List[int]]): List of edges, where each edge is a list of two integers.

    Returns:
        Dict[int, List[int]]: A dictionary representing the graph.

    Raises:
        ValueError: If an edge is invalid.
    """
    graph = {}
    for a, b in edges:
        if not isinstance(a, int) or not isinstance(b, int):
            raise ValueError("Both nodes in an edge must be integers.")
        graph.setdefault(a, [])
        graph.setdefault(b, [])
        graph[a].append(b)
    return graph


def topo_sort(graph: Dict[int, List[int]]) -> List[int]:
    """
    Perform a topological sort on a graph.

    Args:
        graph (Dict[int, List[int]]): A dictionary representing the graph.

    Returns:
        List[int]: A list of nodes in topological order.

    Raises:
        ValueError: If the graph contains a cycle.
    """
    in_degree = {node: 0 for node in graph}
    for node in graph:
        for neighbor in graph[node]:
            in_degree[neighbor] += 1

    queue = [node for node in graph if in_degree[node] == 0]
    topo_order = []

    while queue:
        node = queue.pop(0)
        topo_order.append(node)
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)

    if len(topo_order) != len(graph):
        raise ValueError("The graph contains a cycle.")

    return topo_order

- The function first calculates the in-degree of each node and uses a queue to process nodes with zero in-degree.
- If the length of the topological order is not equal to the number of nodes in the graph, a cycle is detected.
