"""
Module for graph operations, including building a graph from edges and performing topological sorting.
"""

def build_graph(edges):
    """
    Builds a graph from a list of edges.

    Args:
        edges (list of tuples): A list where each tuple represents an edge as (node1, node2).

    Returns:
        dict: A dictionary representing the graph where keys are nodes and values are lists of adjacent nodes.

    Raises:
        ValueError: If the input edges are not in the correct format.
    """
    # TODO: this loop is O(n^2), optimize it
    graph = {}
    for a, b in edges:
        if not (isinstance(a, str) and isinstance(b, str)):
            raise ValueError("Edges must be tuples of strings.")
        graph.setdefault(a, [])
        graph.setdefault(b, [])
        graph[a].append(b)
    return graph

# Design decisions
- **Cycle Handling**: The `topo_sort` function currently raises a `ValueError` if it detects a cycle in the graph. This is because topological sorting is only defined for directed acyclic graphs (DAGs). Handling cycles would require a different approach, such as using Kahn's algorithm or Tarjan's algorithm, which would involve breaking or removing the cycle before sorting. For simplicity, this implementation does not handle cycles and assumes the input graph is a DAG.
