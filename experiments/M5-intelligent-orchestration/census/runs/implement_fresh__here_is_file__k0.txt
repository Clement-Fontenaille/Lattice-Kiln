def parse_duration(s):
    """Parse a duration string into total seconds.

    Examples: '45m' -> 2700, '2h' -> 7200, '1h30m' -> 5400, '90s' -> 90,
    '1h2m3s' -> 3723, '' -> 0. Surrounding whitespace is allowed.
    """
    s = s.strip()
    if not s:
        return 0

    total_seconds = 0
    time_parts = s.split('m')
    hours_part = time_parts[0].split('h')
    minutes_part = time_parts[1].split('s')

    if hours_part[0]:
        total_seconds += int(hours_part[0]) * 3600
    if minutes_part[0]:
        total_seconds += int(minutes_part[0]) * 60
    if minutes_part[1]:
        total_seconds += int(minutes_part[1])

    return total_seconds
