def parse_duration(s):
    """Parse a duration string into total seconds.

    Examples: '45m' -> 2700, '2h' -> 7200, '1h30m' -> 5400, '90s' -> 90,
    '1h2m3s' -> 3723, '' -> 0. Surrounding whitespace is allowed.
    """
    s = s.strip()
    if not s:
        return 0

    total_seconds = 0
    units = {'s': 1, 'm': 60, 'h': 3600}
    parts = s.split('.')

    for part in parts:
        if 'h' in part:
            hours = int(part.split('h')[0])
            total_seconds += hours * units['h']
        elif 'm' in part:
            minutes = int(part.split('m')[0])
            total_seconds += minutes * units['m']
        elif 's' in part:
            seconds = int(part.split('s')[0])
            total_seconds += seconds * units['s']

    return total_seconds
